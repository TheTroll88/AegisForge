import time
from strategies import load_strategies
from sandbox.sandbox_trading import SandboxTrader
from live_trading.live_trading import LiveTrader
from api.binance_client import BinanceClient
from utils.logger import setup_logger
from utils.config_loader import load_config

logger = setup_logger()

def main():
    config = load_config()
    mode = config.get('mode', 'sandbox')
    client = BinanceClient()

    if mode == 'sandbox':
        trader = SandboxTrader(client, config)
    elif mode == 'live':
        trader = LiveTrader(client, config)
    else:
        logger.error(f"Invalid mode: {mode}")
        return

    strategies = load_strategies(config.get('strategies', []))
    logger.info(f"Loaded strategies: {list(strategies.keys())}")

    while True:
        try:
            market_data = client.get_market_data()
            for name, strategy in strategies.items():
                signal = strategy.analyze(market_data)
                if signal:
                    trader.execute_trade(signal)
            time.sleep(config.get('poll_interval', 10))
        except Exception as e:
            logger.error(f"Error in main loop: {e}")

if __name__ == "__main__":
    main()

import logging

def setup_logger():
    logger = logging.getLogger('Kratos')
    logger.setLevel(logging.INFO)
    handler = logging.StreamHandler()
    formatter = logging.Formatter('[%(asctime)s] %(levelname)s:%(message)s')
    handler.setFormatter(formatter)
    logger.addHandler(handler)
    return logger

class MomentumStrategy:
    def __init__(self, config):
        self.config = config

    def analyze(self, market_data):
        closes = market_data['closes']
        if closes[-1] > closes[-2]:
            return {'type': 'BUY', 'confidence': 0.7}
        elif closes[-1] < closes[-2]:
            return {'type': 'SELL', 'confidence': 0.7}
        return None

class LiveTrader:
    def __init__(self, client, config):
        self.client = client
        self.config = config

    def execute_trade(self, signal):
        side = signal['type']
        quantity = self.config.get('trade_amount', 0.001)
        symbol = self.config.get('symbol', 'BTCUSDT')
        self.client.place_order(symbol, side, quantity)

import os
from binance.client import Client
from dotenv import load_dotenv

load_dotenv()

class BinanceClient:
    def __init__(self):
        self.client = Client(os.getenv('BINANCE_API_KEY'), os.getenv('BINANCE_API_SECRET'))

    def get_market_data(self):
        candles = self.client.get_klines(symbol='BTCUSDT', interval=Client.KLINE_INTERVAL_1MINUTE)
        closes = [float(c[4]) for c in candles]
        return {'closes': closes}

    def place_order(self, symbol, side, quantity):
        order = self.client.order_market_buy(symbol=symbol, quantity=quantity) if side == 'BUY' else self.client.order_market_sell(symbol=symbol, quantity=quantity)
        return order

import time
import requests

class Gatekeeper:
    def __init__(self, monitor_url):
        self.monitor_url = monitor_url

    def monitor(self):
        while True:
            try:
                r = requests.get(self.monitor_url)
                if r.status_code != 200:
                    self.send_alert('Service down!')
            except Exception:
                self.send_alert('Error contacting service!')
            time.sleep(60)

    def send_alert(self, message):
        print(f"ALERT: {message}")
